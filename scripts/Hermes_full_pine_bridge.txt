f_array_mean_last(array<float> arr, int n) =>
    int sz = array.size(arr)
    if sz == 0 or n <= 0
        na
    else
        int take = math.min(sz, n)
        float sum = 0.0
        for i = 0 to take - 1
            sum += array.get(arr, sz - 1 - i)
        sum / take

f_array_std_last(array<float> arr, int n) =>
    int sz = array.size(arr)
    if sz <= 1 or n <= 1
        na
    else
        int take = math.min(sz, n)
        float mean = f_array_mean_last(arr, take)
        float ss = 0.0
        for i = 0 to take - 1
            float v = array.get(arr, sz - 1 - i)
            ss += math.pow(v - mean, 2)
        math.sqrt(ss / take)

f_array_corr_last(array<float> arrA, array<float> arrB, int n) =>
    int szA = array.size(arrA)
    int szB = array.size(arrB)
    int take = math.min(math.min(szA, szB), n)
    if take <= 1
        na
    else
        float meanA = 0.0
        float meanB = 0.0
        for i = 0 to take - 1
            meanA += array.get(arrA, szA - 1 - i)
            meanB += array.get(arrB, szB - 1 - i)
        meanA /= take
        meanB /= take

        float cov = 0.0
        float varA = 0.0
        float varB = 0.0
        for i = 0 to take - 1
            float a = array.get(arrA, szA - 1 - i) - meanA
            float b = array.get(arrB, szB - 1 - i) - meanB
            cov += a * b
            varA += a * a
            varB += b * b

        float den = math.sqrt(varA * varB)
        den > 0 ? (cov / den) : na

f_credit_model(float wPut, float wCall) =>
    float widthMin = (wPut > 0 and wCall > 0) ? math.min(wPut, wCall) : na
    if na(widthMin)
        na
    else if useSkewCredit
        float putCreditPct  = creditPct * (1.0 + putSkewBoost)
        float callCreditPct = creditPct * (1.0 - callSkewDiscount)
        float putCredit  = wPut  * putCreditPct
        float callCredit = wCall * callCreditPct
        (putCredit + callCredit) / 2.0
    else
        widthMin * creditPct
f_side_credit_alloc(float totalCreditPts, float wPut, float wCall) =>
    if na(totalCreditPts) or na(wPut) or na(wCall) or wPut <= 0 or wCall <= 0
        [na, na]
    else
        float rawPut  = useSkewCredit ? (wPut  * (1.0 + putSkewBoost))     : wPut
        float rawCall = useSkewCredit ? (wCall * (1.0 - callSkewDiscount)) : wCall
        float rawSum  = rawPut + rawCall
        float putAlloc  = rawSum > 0 ? totalCreditPts * rawPut  / rawSum : totalCreditPts * 0.5
        float callAlloc = rawSum > 0 ? totalCreditPts * rawCall / rawSum : totalCreditPts * 0.5
        [putAlloc, callAlloc]

f_far_otm_score(float cushion, float width, float remSigma, float widthExcessFull, float sigmaExcessFull) =>
    if na(cushion) or na(width) or width <= 0 or cushion <= width
        0.0
    else
        float widthExcess = (cushion / width) - 1.0
        float widthScore = f_clamp(widthExcess / math.max(widthExcessFull, 0.0001), 0.0, 1.0)

        float sigmaScore = widthScore
        if not na(remSigma) and remSigma > 0
            float sigmaExcess = math.max((cushion / remSigma) - 1.0, 0.0)
            sigmaScore := f_clamp(sigmaExcess / math.max(sigmaExcessFull, 0.0001), 0.0, 1.0)

        // Width dominates because it is directly tied to the sold structure.
        0.80 * widthScore + 0.20 * sigmaScore
f_near_short_score(float cushion, float width, float remSigma, float widthStart, float sigmaStart, float convexPow) =>
    if na(cushion) or na(width) or width <= 0
        na
    else
        float widthDen = math.max(width * math.max(widthStart, 0.0001), gridStep)
        float widthCloseness = 1.0 - f_clamp(cushion / widthDen, 0.0, 1.0)

        float sigmaCloseness = widthCloseness
        if not na(remSigma) and remSigma > 0
            float sigmaDen = math.max(remSigma * math.max(sigmaStart, 0.0001), gridStep)
            sigmaCloseness := 1.0 - f_clamp(cushion / sigmaDen, 0.0, 1.0)

        float nearBase = math.max(widthCloseness, sigmaCloseness)
        math.pow(f_clamp(nearBase, 0.0, 1.0), math.max(convexPow, 1.0))

f_late_gamma_score(float minsLeft, float minsTotal, float startRemainFrac, float gammaPow) =>
    if na(minsLeft) or na(minsTotal) or minsTotal <= 0
        na
    else
        float remFrac = f_clamp(minsLeft / minsTotal, 0.0, 1.0)
        float startFrac = f_clamp(startRemainFrac, 0.05, 1.0)
        float progress = remFrac >= startFrac ? 0.0 : ((startFrac - remFrac) / startFrac)
        math.pow(f_clamp(progress, 0.0, 1.0), math.max(gammaPow, 1.0))
f_remaining_extrinsic_frac(float timeLeftFrac, float sideThreat, float riskNorm, float centerQuality, float farOtmScore, float nearShortScore, float surfaceStressNow, float lateGammaScore) =>
    float base = math.pow(f_clamp(nz(timeLeftFrac, 0.0), 0.0, 1.0), markThetaPow)
    float decayProgress = 1.0 - base

    float threatAdj = 1.0 + markThreatBeta * nz(sideThreat, 0.0)
    float riskAdj   = 1.0 + markRiskBeta * nz(riskNorm, 0.0)

    // Calm / safe discounts phase in with elapsed decay progress.
    float centerAdj = 1.0 - markCenterBeta * nz(centerQuality, 0.0) * decayProgress
    float farOtmAdj = 1.0 - markFarOtmBeta * nz(farOtmScore, 0.0) * decayProgress

    // New targeted pre-breach repricing package.
    float nearShortAdj = 1.0 + ((usePreBreachBoost ? markNearShortBeta : 0.0) * nz(nearShortScore, 0.0))
    float surfaceStressAdj = 1.0 + ((usePreBreachBoost ? markSurfaceStressBeta : 0.0) * nz(nearShortScore, 0.0) * nz(surfaceStressNow, 0.0))
    float lateGammaAdj = 1.0 + ((usePreBreachBoost ? markLateGammaBeta : 0.0) * nz(nearShortScore, 0.0) * nz(lateGammaScore, 0.0))

    f_clamp(base * threatAdj * riskAdj * centerAdj * farOtmAdj * nearShortAdj * surfaceStressAdj * lateGammaAdj, markExtrinsicFloorFrac, markExtrinsicCapFrac)

f_profile_center(float legacyCenter, float vwapNow, float priceNow, float sigmaNow, int profileCode) =>
    if na(legacyCenter) or na(sigmaNow) or sigmaNow <= 0
        [legacyCenter, na]
    else
        float out = legacyCenter

        if profileCode == 1
            // RANGE_PIN: pull toward VWAP
            if not na(vwapNow)
                out := f_lerp(legacyCenter, vwapNow, rangePinCenterVWAPWeight)

        else if profileCode == 2 or profileCode == 3
            // BEAR/BULL DRIFT: blend legacy AM center + VWAP + current price + directional bias
            float wV = driftCenterVWAPWeight
            float wP = driftCenterPriceWeight
            float wL = math.max(0.0, 1.0 - wV - wP)
            float wSum = wL + wV + wP

            float base = legacyCenter
            if not na(vwapNow) and not na(priceNow)
                base := (wL * legacyCenter + wV * vwapNow + wP * priceNow) / math.max(wSum, 0.0001)
            else if not na(vwapNow)
                base := f_lerp(legacyCenter, vwapNow, wV / math.max(wV + wL, 0.0001))
            else if not na(priceNow)
                base := f_lerp(legacyCenter, priceNow, wP / math.max(wP + wL, 0.0001))

            float dir = profileCode == 2 ? -1.0 : 1.0
            out := base + dir * driftCenterBiasSigma * sigmaNow

        else if profileCode == 4
            // HAZARD: conservative fallback near VWAP/current price blend
            float hazardBase = legacyCenter
            if not na(vwapNow) and not na(priceNow)
                hazardBase := f_lerp(vwapNow, priceNow, hazardCenterPriceWeight)
            else if not na(vwapNow)
                hazardBase := vwapNow
            else if not na(priceNow)
                hazardBase := priceNow

            out := f_lerp(legacyCenter, hazardBase, 0.50)

        float maxShift = profileCenterMaxShiftSigma * sigmaNow
        out := legacyCenter + f_clamp(out - legacyCenter, -maxShift, maxShift)

        float shiftSigma = (out - legacyCenter) / sigmaNow
        [out, shiftSigma]

f_profile_geom(int code, float strength, float edge, float stress, float skew) =>
    float safeStr = nz(strength, 0.0)
    float safeEdge = nz(edge, 0.0)
    float safeStress = nz(stress, 0.0)
    float safeSkew = nz(skew, 0.0)

    float g = 1.0
    float p = 1.0
    float c = 1.0

    if code == 1
        float tightenScore = safeStr * safeEdge * (1.0 - safeStress)
        g := g * (1.0 - rangeTightenMax * tightenScore)
    else if code == 2
        float tightenScore = safeStr * safeEdge * (1.0 - safeStress)
        g := g * (1.0 - driftGlobalTightenMax * tightenScore)
        p := p * (1.0 + bearPutWidenMax * safeStr * math.max(safeSkew, 0.50))
        c := c * (1.0 - bearCallTightenMax * tightenScore)
    else if code == 3
        float tightenScore = safeStr * safeEdge * (1.0 - safeStress)
        g := g * (1.0 - driftGlobalTightenMax * tightenScore)
        p := p * (1.0 - bullPutTightenMax * tightenScore)
        c := c * (1.0 + bullCallWidenMax * safeStr * 0.50)
    else if code == 4
        float widenScore = safeStr * (0.50 + 0.50 * safeStress)
        g := g * (1.0 + hazardGlobalWidenMax * widenScore)

    g := g * (1.0 - surfaceEdgeTightenMax * safeEdge * (1.0 - safeStress) * (code == 4 ? 0.0 : 1.0))
    g := g * (1.0 + surfaceStressWidenMax * safeStress)
    p := p * (1.0 + surfaceSkewPutProtectMax * safeSkew * ((code == 2 or code == 4) ? 1.0 : 0.25))

    [f_clamp(g, 0.75, 1.35), f_clamp(p, 0.80, 1.60), f_clamp(c, 0.80, 1.35)]

f_build_condor(float center, float sigma, float zBias, float kScaleMult, float ivrv, float extraPutKMult, float extraCallKMult) =>
    if na(center) or na(sigma) or sigma <= 0
        [na, na, na, na, na, na, na, na]
    else
        float kBase = useAutoK ? f_pick_k(ivrv) : kManual_base_eff
        float kUse  = kBase * kScaleMult

        float kPutUse  = useTiltK ? (kUse * f_clamp(1.0 - tiltKbeta * zBias, minKClamp, maxKClamp)) : kUse
        float kCallUse = useTiltK ? (kUse * f_clamp(1.0 + tiltKbeta * zBias, minKClamp, maxKClamp)) : kUse

        kPutUse  := kPutUse  * nz(extraPutKMult, 1.0)
        kCallUse := kCallUse * nz(extraCallKMult, 1.0)
        float wPutUse  = wSigma_eff * f_clamp(1.0 - tiltWingBeta * zBias, 0.50, 2.00)
        float wCallUse = wSigma_eff * f_clamp(1.0 + tiltWingBeta * zBias, 0.50, 2.00)

        float sPutRaw  = center - kPutUse * sigma
        float sCallRaw = center + kCallUse * sigma
        float lPutRaw  = sPutRaw  - wPutUse  * sigma
        float lCallRaw = sCallRaw + wCallUse * sigma

        float sPutR  = f_round_mode(sPutRaw,  gridStep, roundMode, center)
        float sCallR = f_round_mode(sCallRaw, gridStep, roundMode, center)
        float lPutR  = f_round_mode(lPutRaw,  gridStep, "Outward", center)
        float lCallR = f_round_mode(lCallRaw, gridStep, "Outward", center)

        float wPut  = sPutR  - lPutR
        float wCall = lCallR - sCallR
        float widthMin  = (wPut > 0 and wCall > 0) ? math.min(wPut, wCall) : na
        float creditPts = f_credit_model(wPut, wCall)

        [sPutR, lPutR, sCallR, lCallR, widthMin, creditPts, kPutUse, kCallUse]

f_sigma_pts_remaining(float level, float impMove, float minsLeft, float minsTotal) =>
    if na(level) or na(impMove) or impMove <= 0 or na(minsLeft) or na(minsTotal) or minsTotal <= 0
        na
    else
        float remFrac = f_clamp(minsLeft / minsTotal, 0.0, 1.0)
        level * impMove * math.sqrt(remFrac)

f_interval_pop(float center, float sigma, float shortPut, float shortCall) =>
    if na(center) or na(sigma) or sigma <= 0 or na(shortPut) or na(shortCall)
        na
    else
        float zPut  = (center - shortPut) / sigma
        float zCall = (shortCall - center) / sigma
        f_clamp(f_normcdf(zCall) + f_normcdf(zPut) - 1.0, 0.0, 1.0)

f_session_vwap_width_engine(int code) =>
    inSess  = not na(time(timeframe.period, rthSession))
    isOpen  = inSess and not inSess[1]
    var float wSum  = 0.0
    var float wMean = na
    var float wM2   = 0.0
    var float vwap  = na
    var float ssd   = na
    if isOpen
        wSum := 0.0, wMean := na, wM2 := 0.0, vwap := na, ssd := na
    if inSess
        float w = na(volume) or volume <= 0 ? (useVolFallback ? 1.0 : na) : volume
        if not na(w)
            float x = f_src(vwapSrcOpt)
            if na(wMean)
                wSum := w, wMean := x, wM2 := 0.0
            else
                float sumNew = wSum + w
                float delta  = x - wMean
                float meanNew= wMean + (w * delta) / sumNew
                wM2 := wM2 + w * delta * (x - meanNew), wSum := sumNew, wMean := meanNew
            vwap := wMean, ssd := wSum > 0 ? math.sqrt(wM2 / wSum) : na
    atrv  = ta.atr(atrLen)
    width = code == 1 ? atrv : ssd
    [vwap, width]

f_opening_range(float mins) =>

    var float hi = na
    var float lo = na
    var bool  done = false
    if isOpen
        hi := high
        lo := low
        done := false
    if inSess and not done
        if time_close <= endMs
            hi := math.max(hi, high)
            lo := math.min(lo, low)
        else
            done := true
    [hi, lo, done]

                        eng_wSum := sumNew
                        eng_wMean := meanNew

                eng_sSD := eng_wSum > 0 ? math.sqrt(eng_wM2 / eng_wSum) : na

                float sigmaRemReplay = f_sigma_pts_remaining(src_i, impMv, minsLeftReplay, rthSessionMinutes)
                float widthFloorReplay = math.max(gridStep, math.max(nz(atrVal, gridStep) * 0.25, nz(sigmaRemReplay, 0.0) * 0.05))

                    eng_distTrendEma := na(eng_distTrendEma) ? dist_i : (eng_distTrendEma + aTrendReplay * (dist_i - eng_distTrendEma))
                    eng_distLocalVar := na(eng_distLocalVar) ? (dist_i * dist_i) : (eng_distLocalVar + aLocalReplay * (dist_i * dist_i - eng_distLocalVar))

                    float widthLocalReplay = not na(prevLocalVar) ? math.max(math.sqrt(math.max(prevLocalVar, 0.0)), widthFloorReplay) : widthFloorReplay
                    float widthSlowReplay  = not na(prevWidthSlow) ? math.max(prevWidthSlow, widthFloorReplay) : widthFloorReplay
                    float widthHybridReplay = math.sqrt(0.70 * widthLocalReplay * widthLocalReplay + 0.30 * widthSlowReplay * widthSlowReplay)

                    eng_zNow := (not na(eng_sVWAP) and not na(widthSlowReplay) and widthSlowReplay > 0) ? ((src_i - eng_sVWAP) / widthSlowReplay) : na
                    eng_zTrigger := widthLocalReplay > 0 ? (dist_i / widthLocalReplay) : na
                    eng_zShape := widthHybridReplay > 0 ? f_clamp(eng_distShapeEma / widthHybridReplay, -1.50, 1.50) : na
                    eng_zTrend := widthSlowReplay > 0 ? (eng_distTrendEma / widthSlowReplay) : na
                    eng_zVWAPRecenter := widthHybridReplay > 0 ? (dist_i / widthHybridReplay) : na

                    array.push(engDistHist, dist_i)
                    if array.size(engDistHist) > recenterLB + 1
                        array.shift(engDistHist)

                    eng_distAbsImprovement := array.size(engDistHist) > recenterLB ? (math.abs(array.get(engDistHist, 0)) - math.abs(dist_i)) : na
                    eng_nearVWAPRecenter := math.abs(dist_i) <= nearVwapFrac * widthHybridReplay
                    eng_movingTowardVWAP := not na(eng_distAbsImprovement) and eng_distAbsImprovement >= 0.0

    if inRTH and not na(eng_srcVW)
        sVWAP_eff := eng_sVWAP
        baseWidth_eff := bandMethod == "ATR" ? atrVal : eng_sSD
        srcVW_eff := eng_srcVW

        zNow := eng_zNow
        zTrigger := eng_zTrigger

f_breach_loss(float slX, float creditPtsEntry, float shortStrike, float observedPx, float sideWidth, bool isPut) =>
    float baseLoss = slX * creditPtsEntry
    if useNonLinearBreach and not na(sideWidth) and sideWidth > 0
        float overshoot = isPut ? math.max(0.0, shortStrike - observedPx) : math.max(0.0, observedPx - shortStrike)
        float overshootRatio = overshoot / sideWidth
        float severityAdj = 1.0 + breachSeverityMult * overshootRatio
        baseLoss * severityAdj
    else
        baseLoss
